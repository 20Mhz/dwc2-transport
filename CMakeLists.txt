cmake_minimum_required(VERSION 3.16)
project(dwc2_transport VERSION 1.2.0 LANGUAGES C)
include(GNUInstallDirs)
include(CTest)
if(APPLE)
  set(CMAKE_INSTALL_RPATH "@loader_path/../${CMAKE_INSTALL_LIBDIR}")
else()
  set(CMAKE_INSTALL_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR}")
endif()
option(BUILD_SHARED_LIBS "Build shared library" OFF)
option(DT_SANITIZERS "Enable address/undefined sanitizers" OFF)
add_library(dwc2_transport src/common/core.c src/common/async.c src/linux/usbfs.c src/mock/mock.c)
target_compile_features(dwc2_transport PUBLIC c_std_11)
target_compile_definitions(dwc2_transport PRIVATE _GNU_SOURCE)
target_include_directories(dwc2_transport PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(dwc2_transport PRIVATE -Wall -Wextra -Werror -Wshadow -Wstrict-prototypes -Wmissing-prototypes)
  add_compile_options( -Wall -Wextra -Werror -Wshadow -Wstrict-prototypes -Wmissing-prototypes)
endif()
set_target_properties(dwc2_transport PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 1 C_VISIBILITY_PRESET hidden)
if(DT_SANITIZERS)
  target_compile_options(dwc2_transport PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(dwc2_transport PUBLIC -fsanitize=address,undefined)
endif()
foreach(t usb-discover usb-inspect usb-diagnostics)
  add_executable(${t} tools/cli.c tools/hostlab/jsonl.c tools/hostlab/clock.c tools/diagnostics/descriptors.c)
endforeach()
add_executable(read-descriptor examples/read_descriptor.c)
foreach(t usb-discover usb-inspect usb-diagnostics read-descriptor)
  target_link_libraries(${t} PRIVATE dwc2_transport)
endforeach()
if(BUILD_TESTING)
  add_executable(test-hardware tests/hardware/regression.c)
  target_link_libraries(test-hardware PRIVATE dwc2_transport)
  add_executable(test-descriptors tests/unit/test_descriptors.c tools/diagnostics/descriptors.c)
  target_link_libraries(test-descriptors PRIVATE dwc2_transport)
  add_test(NAME descriptors COMMAND test-descriptors)
  add_executable(test-inspection tests/unit/test_inspection.c tools/diagnostics/descriptors.c tools/hostlab/jsonl.c tools/hostlab/clock.c)
  target_link_libraries(test-inspection PRIVATE dwc2_transport)
  add_executable(test-core tests/unit/test_core.c)
  target_link_libraries(test-core PRIVATE dwc2_transport)
  add_test(NAME core COMMAND test-core)
  add_executable(test-async tests/unit/test_async.c)
  target_link_libraries(test-async PRIVATE dwc2_transport)
  add_test(NAME async COMMAND test-async)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND NOT BUILD_SHARED_LIBS)
    add_executable(test-linux tests/integration/test_linux.c)
    target_compile_definitions(test-linux PRIVATE _GNU_SOURCE)
    target_link_libraries(test-linux PRIVATE dwc2_transport)
    target_link_options(test-linux PRIVATE -Wl,--wrap=ioctl -Wl,--wrap=poll -Wl,--wrap=clock_gettime -Wl,--wrap=close)
    add_test(NAME linux-contracts COMMAND test-linux)
  endif()
  add_executable(test-replay tests/unit/test_replay.c)
  target_link_libraries(test-replay PRIVATE dwc2_transport)
  add_test(NAME replay COMMAND test-replay ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/hostlab-ch340.replay ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/invalid.replay)
  set_tests_properties(async PROPERTIES TIMEOUT 20)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_FOUND AND NOT CMAKE_CROSSCOMPILING)
    add_test(NAME cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/test_cli.py ${CMAKE_CURRENT_BINARY_DIR})
    add_test(NAME evidence COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/test_evidence.py)
    add_test(NAME observer COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/test_observer.py)
    add_test(NAME recording COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/test_recording.py)
    add_test(NAME timeline COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/test_timeline.py)
    add_test(NAME release COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/test_release.py)
    add_test(NAME privacy COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/test_privacy.py)
    add_test(NAME database COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/scripts/generate-device-db.py --check)
    add_test(NAME apple-capture COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/test_apple_capture.py)
  endif()
  add_test(NAME descriptor-mock COMMAND read-descriptor --mock)
endif()
install(TARGETS dwc2_transport usb-discover usb-inspect usb-diagnostics read-descriptor EXPORT DWC2TransportTargets
  ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/dwc2_transport DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT DWC2TransportTargets FILE DWC2TransportConfig.cmake NAMESPACE DWC2:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/DWC2Transport)
include(CMakePackageConfigHelpers)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/DWC2TransportConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
file(RELATIVE_PATH DT_PC_PREFIX "/${CMAKE_INSTALL_LIBDIR}/pkgconfig" "/")
configure_file(cmake/dwc2-transport.pc.in dwc2-transport.pc @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/DWC2TransportConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/DWC2Transport)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/dwc2-transport.pc DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)
install(FILES LICENSE NOTICE DESTINATION ${CMAKE_INSTALL_DATADIR}/dwc2-transport)

install(PROGRAMS tools/usb-observe tools/usb-evidence-verify tools/usb-annotate DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY python/dwc2_diag DESTINATION ${CMAKE_INSTALL_DATADIR}/dwc2-transport/python FILES_MATCHING PATTERN "*.py")
install(FILES data/device-identification-v1.json DESTINATION ${CMAKE_INSTALL_DATADIR}/dwc2-transport/data)
